boolean binsearch (int[] haystack, int needle){
	int highest = haystack.length-1;
	int lowest = 0;
	while (lowest<=highest){
		int middle = highest + lowest / 2;
		if (haystack[middle]==needle){
			return true;
		}else if(haystack[middle]<needle){
			lowest = middle + 1;
		}else{
			highest  = middle -1;
		}
	}
	return false;
}

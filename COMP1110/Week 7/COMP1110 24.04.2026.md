public sealed interface Entry permits FileEntry, DirEntry {

    long size();

    String name();

    static Entry entryForFile(File f){ //what does static mean? // this is associated with the class, rather than a value
    //this function entryForFile will return 
            if(f.isDirectory()){
                List<Entry> contents = new ArrayList<>(); 
                for (File g: f.listFiles()){
                    contents.add(Entry.entryForFile(g));
                }
                Collections.sort(contents); //uses tim sort
                return new DirEntry(f.getName(), contents);
            }else{
                return new FileEntry(f.getName(), f.length());
            }
    }

    default int compareTo(Entry e){ //default method, unless implemented in class, this will be used

        return Long.compare(e.size(), this.size());

    }

}

record FileEntry(String name, long size) implements Entry{
}

  

record DirEntry(String name, List<Entry> files) implements Entry {

 public long size(){
    long size = 0;
    for(Entry entry: files){
    size = size + entry.size();
    }
	return size;    
 }  
}

```java
boolean noSpecialCharacters(String pwd){
	int pos = 0;
	//iteration using while loop
	//often, using < would be a better option 
	while (pos != pwd.length()) {
		char ch = pwd.charAt(pos);
		if (!Character.isLetterOrDigit(ch)){
		return false;
		}
		pos = pos + 1;
	}
	return true;
}
```

```java
boolean containsCat(String pwd){
//go through password, if we see c a and t in a sequence, return true 
	int i = 0
	while(i < pwd.length() - 2){
	char ith = pwd.charAt(i)
	char ipoth = pwd.charAt(i+1)
	char iptth = pwd.charAt(i+2)
	if (ith == 'c' && ipoth == 'a' && iptth == 't'{
		return true;
	}

	}
	return false;
```
